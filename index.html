1. Supabase PostgreSQL — 12 Tabel & Index (`001_schema_12_tables.sql`) 
-- ============================================================================
-- SIPUSTADIG (SIRKULIB) v3.0 — SUPABASE POSTGRESQL SCHEMA (12 TABEL + INDEXES)
-- Sesuai PRD.md v3.0 & DESIGN.md
-- ============================================================================

-- 1. TABEL USERS (Sinkron dengan auth.users Supabase)
CREATE TABLE IF NOT EXISTS public.users (
  id BIGSERIAL PRIMARY KEY,
  auth_uid UUID UNIQUE,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) UNIQUE NOT NULL,
  password VARCHAR(255),
  role VARCHAR(30) NOT NULL DEFAULT 'student' CHECK (role IN ('admin', 'librarian', 'principal', 'teacher', 'student')),
  status VARCHAR(20) NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'inactive')),
  last_login_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. TABEL MEMBERS (Anggota Perpustakaan: Siswa, Guru, Staf)
CREATE TABLE IF NOT EXISTS public.members (
  id BIGSERIAL PRIMARY KEY,
  user_id BIGINT UNIQUE REFERENCES public.users(id) ON DELETE SET NULL,
  member_number VARCHAR(50) UNIQUE NOT NULL,
  type VARCHAR(20) NOT NULL DEFAULT 'student' CHECK (type IN ('student', 'teacher', 'staff')),
  name VARCHAR(255) NOT NULL,
  identity_number VARCHAR(50) NOT NULL,
  class_name VARCHAR(50),
  major VARCHAR(100),
  email VARCHAR(255),
  phone VARCHAR(20),
  address TEXT,
  photo_path VARCHAR(255),
  status VARCHAR(20) NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'inactive')),
  notes TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 3. TABEL BOOK_CATEGORIES (Kategori Buku)
CREATE TABLE IF NOT EXISTS public.book_categories (
  id BIGSERIAL PRIMARY KEY,
  name VARCHAR(100) UNIQUE NOT NULL,
  slug VARCHAR(100) UNIQUE NOT NULL,
  description TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 4. TABEL BOOKS (Bibliografi / Katalog Buku)
CREATE TABLE IF NOT EXISTS public.books (
  id BIGSERIAL PRIMARY KEY,
  category_id BIGINT REFERENCES public.book_categories(id) ON DELETE SET NULL,
  title VARCHAR(255) NOT NULL,
  isbn VARCHAR(20),
  author VARCHAR(255) NOT NULL,
  publisher VARCHAR(255),
  published_year INT,
  language VARCHAR(50) NOT NULL DEFAULT 'Indonesia',
  pages INT,
  description TEXT,
  cover_path VARCHAR(255),
  tags JSONB DEFAULT '[]'::jsonb,
  status VARCHAR(20) NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'inactive')),
  created_by BIGINT REFERENCES public.users(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 5. TABEL BOOK_COPIES (Eksemplar Fisik & Barcode)
CREATE TABLE IF NOT EXISTS public.book_copies (
  id BIGSERIAL PRIMARY KEY,
  book_id BIGINT NOT NULL REFERENCES public.books(id) ON DELETE CASCADE,
  copy_code VARCHAR(50) UNIQUE NOT NULL,
  barcode_value VARCHAR(100) UNIQUE NOT NULL,
  shelf_location VARCHAR(100),
  condition VARCHAR(30) NOT NULL DEFAULT 'good' CHECK (condition IN ('good', 'fair', 'lightly_damaged', 'heavily_damaged', 'lost')),
  status VARCHAR(30) NOT NULL DEFAULT 'available' CHECK (status IN ('available', 'borrowed', 'damaged', 'lost', 'repair', 'archived')),
  source VARCHAR(30) NOT NULL DEFAULT 'purchase' CHECK (source IN ('purchase', 'grant', 'donation', 'government_aid', 'other')),
  acquisition_date DATE,
  price NUMERIC(10,2) DEFAULT 50000.00,
  notes TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 6. TABEL LOAN_TRANSACTIONS (Header Transaksi Peminjaman)
CREATE TABLE IF NOT EXISTS public.loan_transactions (
  id BIGSERIAL PRIMARY KEY,
  loan_number VARCHAR(50) UNIQUE NOT NULL,
  member_id BIGINT NOT NULL REFERENCES public.members(id) ON DELETE RESTRICT,
  borrowed_by BIGINT REFERENCES public.users(id) ON DELETE SET NULL,
  borrowed_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  due_at DATE NOT NULL,
  returned_at TIMESTAMPTZ,
  status VARCHAR(20) NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'returned', 'overdue', 'lost', 'cancelled')),
  notes TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 7. TABEL FINES (Denda Keterlambatan, Kerusakan & Kehilangan)
CREATE TABLE IF NOT EXISTS public.fines (
  id BIGSERIAL PRIMARY KEY,
  member_id BIGINT NOT NULL REFERENCES public.members(id) ON DELETE RESTRICT,
  loan_item_id BIGINT,
  type VARCHAR(20) NOT NULL DEFAULT 'late' CHECK (type IN ('late', 'damaged', 'lost', 'manual')),
  amount NUMERIC(10,2) NOT NULL DEFAULT 0.00,
  paid_amount NUMERIC(10,2) NOT NULL DEFAULT 0.00,
  status VARCHAR(20) NOT NULL DEFAULT 'unpaid' CHECK (status IN ('unpaid', 'partial', 'paid', 'waived')),
  notes TEXT,
  waived_reason TEXT,
  created_by BIGINT REFERENCES public.users(id) ON DELETE SET NULL,
  paid_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 8. TABEL LOAN_ITEMS (Detail Eksemplar yang Dipinjam per Transaksi)
CREATE TABLE IF NOT EXISTS public.loan_items (
  id BIGSERIAL PRIMARY KEY,
  loan_transaction_id BIGINT NOT NULL REFERENCES public.loan_transactions(id) ON DELETE CASCADE,
  book_copy_id BIGINT NOT NULL REFERENCES public.book_copies(id) ON DELETE RESTRICT,
  borrowed_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  due_at DATE NOT NULL,
  returned_at TIMESTAMPTZ,
  return_condition VARCHAR(30) CHECK (return_condition IN ('good', 'fair', 'lightly_damaged', 'heavily_damaged', 'lost')),
  extension_count INT NOT NULL DEFAULT 0 CHECK (extension_count <= 2),
  status VARCHAR(20) NOT NULL DEFAULT 'borrowed' CHECK (status IN ('borrowed', 'returned', 'overdue', 'lost', 'damaged')),
  fine_id BIGINT REFERENCES public.fines(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Tambahkan FK loan_item_id pada tabel fines
ALTER TABLE public.fines
  DROP CONSTRAINT IF EXISTS fk_fines_loan_item,
  ADD CONSTRAINT fk_fines_loan_item FOREIGN KEY (loan_item_id) REFERENCES public.loan_items(id) ON DELETE SET NULL;

-- 9. TABEL FINE_PAYMENTS (Riwayat Pembayaran Denda)
CREATE TABLE IF NOT EXISTS public.fine_payments (
  id BIGSERIAL PRIMARY KEY,
  fine_id BIGINT NOT NULL REFERENCES public.fines(id) ON DELETE CASCADE,
  amount NUMERIC(10,2) NOT NULL,
  payment_method VARCHAR(20) NOT NULL DEFAULT 'cash' CHECK (payment_method IN ('cash', 'transfer', 'other')),
  paid_by BIGINT REFERENCES public.users(id) ON DELETE SET NULL,
  paid_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  notes TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 10. TABEL LIBRARY_SETTINGS (Konfigurasi Aturan Perpustakaan)
CREATE TABLE IF NOT EXISTS public.library_settings (
  id BIGSERIAL PRIMARY KEY,
  key VARCHAR(100) UNIQUE NOT NULL,
  value TEXT NOT NULL,
  type VARCHAR(20) NOT NULL DEFAULT 'string' CHECK (type IN ('string', 'number', 'boolean', 'json')),
  description TEXT,
  "group" VARCHAR(50) NOT NULL DEFAULT 'general',
  updated_by BIGINT REFERENCES public.users(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 11. TABEL IMPORTS (Riwayat Import Excel/Spreadsheet via GAS)
CREATE TABLE IF NOT EXISTS public.imports (
  id BIGSERIAL PRIMARY KEY,
  type VARCHAR(30) NOT NULL CHECK (type IN ('member', 'book', 'book_copy')),
  file_path VARCHAR(255) NOT NULL,
  status VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'processing', 'completed', 'failed')),
  total_rows INT NOT NULL DEFAULT 0,
  success_rows INT NOT NULL DEFAULT 0,
  failed_rows INT NOT NULL DEFAULT 0,
  error_report_path VARCHAR(255),
  created_by BIGINT REFERENCES public.users(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 12. TABEL AUDIT_LOGS (Pencatatan Aktivitas Sistem)
CREATE TABLE IF NOT EXISTS public.audit_logs (
  id BIGSERIAL PRIMARY KEY,
  user_id BIGINT REFERENCES public.users(id) ON DELETE SET NULL,

  auditable_type VARCHAR(100) NOT NULL,
  auditable_id BIGINT,
  old_values JSONB,
  new_values JSONB,
  ip_address VARCHAR(45),
  user_agent VARCHAR(255),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

2. Supabase PostgreSQL — Fungsi RPC Atomic Pinjam, Kembali & Perpanjang (`003_rpc_transactions.sql`) 
-- ============================================================================
-- SIPUSTADIG (SIRKULIB) v3.0 — POSTGRESQL ATOMIC RPC FUNCTIONS (PL/pgSQL)
-- Mengimplementasikan seluruh aturan bisnis PRD.md Bagian 5:
-- - Siswa: maks 2 eksemplar, durasi 7 hari
-- - Guru: maks 5 eksemplar, durasi 14 hari
-- - Staf: maks 3 eksemplar, durasi 7 hari
-- - Maksimal perpanjangan: 2x (hanya jika belum terlambat)
-- - Denda terlambat: Rp1.000/hari
-- - Denda hilang: Harga buku + 50%
-- ============================================================================

-- 1. FUNGSI TRANSAKSI PEMINJAMAN (process_loan_transaction)
CREATE OR REPLACE FUNCTION public.process_loan_transaction(
  p_member_number VARCHAR,
  p_copy_code VARCHAR,
  p_staff_user_id BIGINT DEFAULT 1,
  p_notes TEXT DEFAULT NULL
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_member RECORD;
  v_copy RECORD;
  v_unpaid_fine NUMERIC(10,2);
  v_active_loans INT;
  v_max_loan INT;
  v_loan_days INT;
  v_due_date DATE;
  v_loan_number VARCHAR(50);
  v_tx_id BIGINT;
  v_item_id BIGINT;
BEGIN
  -- 1. Cek Anggota
  SELECT * INTO v_member FROM public.members WHERE member_number = p_member_number LIMIT 1;
  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'message', 'Nomor anggota tidak ditemukan.');
  END IF;

  IF v_member.status <> 'active' THEN
    RETURN jsonb_build_object('success', false, 'message', 'Anggota berstatus nonaktif dan tidak dapat meminjam buku.');
  END IF;

  -- 2. Cek Denda Belum Lunas
  SELECT COALESCE(SUM(amount - paid_amount), 0) INTO v_unpaid_fine
  FROM public.fines
  WHERE member_id = v_member.id AND status IN ('unpaid', 'partial');

  IF v_unpaid_fine > 0 THEN
    RETURN jsonb_build_object(
      'success', false,
      'message', format('Anggota masih memiliki denda belum lunas sebesar Rp%s.', to_char(v_unpaid_fine, 'FM999G999G999'))
    );
  END IF;

  -- 3. Tentukan Kuota & Durasi sesuai Tipe Anggota (PRD Bagian 5.1)
  IF v_member.type = 'teacher' THEN
    v_max_loan := 5;
    v_loan_days := 14;
  ELSIF v_member.type = 'staff' THEN
    v_max_loan := 3;
    v_loan_days := 7;
  ELSE
    -- Default: student (Siswa)
    v_max_loan := 2;
    v_loan_days := 7;
  END IF;

  -- Hitung pinjaman aktif saat ini
  SELECT COUNT(*) INTO v_active_loans
  FROM public.loan_items li
  JOIN public.loan_transactions lt ON lt.id = li.loan_transaction_id
  WHERE lt.member_id = v_member.id AND li.status IN ('borrowed', 'overdue');

  IF v_active_loans >= v_max_loan THEN
    RETURN jsonb_build_object(
      'success', false,
      'message', format('Jumlah pinjaman anggota sudah mencapai batas maksimal (%s eksemplar untuk %s).', v_max_loan, v_member.type)
    );
  END IF;

  -- 4. Cek Ketersediaan Eksemplar
  SELECT bc.*, b.title AS book_title INTO v_copy
  FROM public.book_copies bc
  JOIN public.books b ON b.id = bc.book_id
  WHERE bc.copy_code = p_copy_code OR bc.barcode_value = p_copy_code
  LIMIT 1;

  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'message', 'Kode eksemplar atau barcode buku tidak ditemukan.');
  END IF;

  IF v_copy.status <> 'available' THEN
    RETURN jsonb_build_object('success', false, 'message', 'Buku ini sedang dipinjam atau tidak tersedia.');
  END IF;

  -- 5. Buat Nomor Transaksi & Hitung Jatuh Tempo
  v_due_date := CURRENT_DATE + v_loan_days;
  v_loan_number := 'TRX-' || to_char(NOW(), 'YYYYMMDD') || '-' || LPAD((FLOOR(RANDOM() * 9000) + 1000)::TEXT, 4, '0');

  INSERT INTO public.loan_transactions (loan_number, member_id, borrowed_by, borrowed_at, due_at, status, notes)
  VALUES (v_loan_number, v_member.id, p_staff_user_id, NOW(), v_due_date, 'active', p_notes)
  RETURNING id INTO v_tx_id;

  INSERT INTO public.loan_items (loan_transaction_id, book_copy_id, borrowed_at, due_at, status)
  VALUES (v_tx_id, v_copy.id, NOW(), v_due_date, 'borrowed')
  RETURNING id INTO v_item_id;

  -- Update status eksemplar menjadi borrowed
  UPDATE public.book_copies
  SET status = 'borrowed', updated_at = NOW()
  WHERE id = v_copy.id;

  -- Catat Audit Log
  INSERT INTO public.audit_logs (user_id, action, auditable_type, auditable_id, new_values)
  VALUES (
    p_staff_user_id,
    'loan',
    'loan_transactions',
    v_tx_id,
    jsonb_build_object('loan_number', v_loan_number, 'member_number', v_member.member_number, 'copy_code', v_copy.copy_code, 'due_at', v_due_date)
  );

  RETURN jsonb_build_object(
    'success', true,
    'message', 'Peminjaman berhasil disimpan.',
    'data', jsonb_build_object(
      'loan_transaction_id', v_tx_id,
      'loan_item_id', v_item_id,
      'loan_number', v_loan_number,
      'member_name', v_member.name,
      'book_title', v_copy.book_title,
      'copy_code', v_copy.copy_code,
      'due_at', v_due_date
    )
  );
END;
$$;

-- 2. FUNGSI TRANSAKSI PENGEMBALIAN (process_return_transaction)
CREATE OR REPLACE FUNCTION public.process_return_transaction(
  p_copy_code VARCHAR,
  p_return_condition VARCHAR DEFAULT 'good',
  p_staff_user_id BIGINT DEFAULT 1,
  p_manual_damage_fine NUMERIC DEFAULT 10000.00
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_item RECORD;
  v_days_late INT;
  v_daily_rate NUMERIC(10,2) := 1000.00;
  v_fine_amount NUMERIC(10,2) := 0.00;
  v_fine_type VARCHAR(20) := 'late';
  v_fine_id BIGINT := NULL;
  v_new_copy_status VARCHAR(30) := 'available';
  v_new_item_status VARCHAR(20) := 'returned';
BEGIN
  -- Cari pinjaman aktif berdasarkan copy_code atau barcode_value
  SELECT
    li.id AS loan_item_id,
    li.loan_transaction_id,
    li.due_at,
    lt.loan_number,
    lt.member_id,
    m.name AS member_name,
    bc.id AS book_copy_id,
    bc.copy_code,
    COALESCE(bc.price, 50000.00) AS book_price,
    b.title AS book_title
  INTO v_item
  FROM public.loan_items li
  JOIN public.loan_transactions lt ON lt.id = li.loan_transaction_id
  JOIN public.members m ON m.id = lt.member_id
  JOIN public.book_copies bc ON bc.id = li.book_copy_id
  JOIN public.books b ON b.id = bc.book_id
  WHERE (bc.copy_code = p_copy_code OR bc.barcode_value = p_copy_code)
    AND li.status IN ('borrowed', 'overdue')
  ORDER BY li.borrowed_at DESC
  LIMIT 1;

  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'message', 'Tidak ditemukan pinjaman aktif untuk kode eksemplar tersebut.');
  END IF;

  -- Hitung keterlambatan hari kalender
  v_days_late := GREATEST(0, CURRENT_DATE - v_item.due_at);
  IF v_days_late > 0 THEN
    v_fine_amount := v_days_late * v_daily_rate;
    v_fine_type := 'late';
  END IF;

  -- Evaluasi kondisi eksemplar saat dikembalikan (PRD Bagian 5.2)
  IF p_return_condition = 'heavily_damaged' THEN
    v_new_copy_status := 'damaged';
    v_new_item_status := 'damaged';
    v_fine_amount := v_fine_amount + COALESCE(p_manual_damage_fine, 10000.00);
    v_fine_type := 'damaged';
  ELSIF p_return_condition = 'lost' THEN
    v_new_copy_status := 'lost';
    v_new_item_status := 'lost';
    -- Denda hilang = Harga buku + 50%
    v_fine_amount := v_fine_amount + (v_item.book_price * 1.5);
    v_fine_type := 'lost';
  ELSE
    v_new_copy_status := 'available';
    v_new_item_status := 'returned';
  END IF;

  -- Buat record denda jika ada tagihan
  IF v_fine_amount > 0 THEN
    INSERT INTO public.fines (member_id, loan_item_id, type, amount, paid_amount, status, notes, created_by)
    VALUES (
      v_item.member_id,
      v_item.loan_item_id,
      v_fine_type,
      v_fine_amount,
      0.00,
      'unpaid',
      format('Denda otomatis (%s): Terlambat %s hari, Kondisi: %s', v_fine_type, v_days_late, p_return_condition),
      p_staff_user_id
    )
    RETURNING id INTO v_fine_id;
  END IF;

  -- Update loan_items
  UPDATE public.loan_items
  SET
    returned_at = NOW(),
    return_condition = p_return_condition,
    status = v_new_item_status,
    fine_id = v_fine_id,
    updated_at = NOW()
  WHERE id = v_item.loan_item_id;

  -- Update loan_transactions
  UPDATE public.loan_transactions
  SET
    returned_at = NOW(),
    status = CASE WHEN v_new_item_status = 'lost' THEN 'lost' ELSE 'returned' END,
    updated_at = NOW()
  WHERE id = v_item.loan_transaction_id;

  -- Update book_copies
  UPDATE public.book_copies
  SET
    status = v_new_copy_status,
    condition = p_return_condition,
    updated_at = NOW()
  WHERE id = v_item.book_copy_id;

  -- Catat Audit Log
  INSERT INTO public.audit_logs (user_id, action, auditable_type, auditable_id, new_values)
  VALUES (
    p_staff_user_id,
    'return',
    'loan_items',
    v_item.loan_item_id,
    jsonb_build_object(
      'copy_code', v_item.copy_code,
      'return_condition', p_return_condition,
      'days_late', v_days_late,
      'fine_amount', v_fine_amount
    )
  );

  RETURN jsonb_build_object(
    'success', true,
    'message', 'Pengembalian berhasil diproses.',
    'data', jsonb_build_object(
      'loan_number', v_item.loan_number,
      'member_name', v_item.member_name,
      'book_title', v_item.book_title,
      'days_late', v_days_late,
      'fine_amount', v_fine_amount,
      'return_condition', p_return_condition
    )
  );
END;
$$;

-- 3. FUNGSI PERPANJANGAN PEMINJAMAN (extend_loan_item)
CREATE OR REPLACE FUNCTION public.extend_loan_item(
  p_copy_code VARCHAR,
  p_staff_user_id BIGINT DEFAULT 1
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_item RECORD;
  v_add_days INT;
  v_new_due DATE;
BEGIN
  SELECT
    li.*,
    lt.member_id,
    m.type AS member_type
  INTO v_item
  FROM public.loan_items li
  JOIN public.loan_transactions lt ON lt.id = li.loan_transaction_id
  JOIN public.members m ON m.id = lt.member_id
  JOIN public.book_copies bc ON bc.id = li.book_copy_id
  WHERE (bc.copy_code = p_copy_code OR bc.barcode_value = p_copy_code)
    AND li.status = 'borrowed'
  LIMIT 1;

  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'message', 'Pinjaman aktif tidak ditemukan.');
  END IF;

  IF CURRENT_DATE > v_item.due_at THEN
    RETURN jsonb_build_object('success', false, 'message', 'Pinjaman yang sudah melewati jatuh tempo (terlambat) tidak dapat diperpanjang.');
  END IF;

  IF v_item.extension_count >= 2 THEN
    RETURN jsonb_build_object('success', false, 'message', 'Batas perpanjangan maksimal (2x) telah tercapai.');
  END IF;

  v_add_days := CASE WHEN v_item.member_type = 'teacher' THEN 14 ELSE 7 END;
  v_new_due := v_item.due_at + v_add_days;

  UPDATE public.loan_items
  SET due_at = v_new_due, extension_count = extension_count + 1, updated_at = NOW()
  WHERE id = v_item.id;

  UPDATE public.loan_transactions
  SET due_at = v_new_due, updated_at = NOW()
  WHERE id = v_item.loan_transaction_id;

  RETURN jsonb_build_object(
    'success', true,
    'message', format('Masa peminjaman berhasil diperpanjang hingga %s (Perpanjangan ke-%s/2).', v_new_due, v_item.extension_count + 1),
    'data', jsonb_build_object('new_due_at', v_new_due, 'extension_count', v_item.extension_count + 1)
  );
END;
$$;

/**
 * ============================================================================
 * SIPUSTADIG (SIRKULIB) v3.0 — BACKEND GOOGLE APPS SCRIPT (Code.gs)
 * Arsitektur: Frontend Vercel + Backend GAS + Database PostgreSQL Supabase
 * ============================================================================
 * Cara Setup di script.google.com:
 * 1. Buka Project Settings > Script Properties, tambahkan:
 *    - SUPABASE_URL              : https://[PROJECT_REF].supabase.co
 *    - SUPABASE_SERVICE_ROLE_KEY : eyJhbGciOiJIUzI1NiIsInR5cCI6...
 *    - DRIVE_COVERS_FOLDER_ID    : ID Folder Google Drive untuk Cover/PDF/Import

 * 2. Klik Deploy > New deployment > Select type: Web app
 *    - Execute as: Me (Akun Google Sekolah)
 *    - Who has access: Anyone
 * 3. Salin Web App URL (/exec) ke environment variable Vercel (VITE_GAS_WEBAPP_URL).
 */

function getConfig_() {
  const props = PropertiesService.getScriptProperties();
  return {
    supabaseUrl: props.getProperty('SUPABASE_URL') || '',
    supabaseKey: props.getProperty('SUPABASE_SERVICE_ROLE_KEY') || '',
    driveFolderId: props.getProperty('DRIVE_COVERS_FOLDER_ID') || '',
    reportSheetId: props.getProperty('REPORTS_SPREADSHEET_ID') || ''
  };
}

/**
 * Helper untuk membuat respons JSON standar sesuai DESIGN.md
 */
function jsonResponse_(payload) {
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Helper jembatan komunikasi REST ke Supabase PostgreSQL (PostgREST)
 */
function supabaseFetch_(endpoint, method, bodyObj, preferHeader) {
  const cfg = getConfig_();
  if (!cfg.supabaseUrl || !cfg.supabaseKey) {
    throw new Error('SUPABASE_URL atau SUPABASE_SERVICE_ROLE_KEY belum diatur di Script Properties.');
  }
  const url = cfg.supabaseUrl.replace(/\/$/, '') + '/rest/v1/' + endpoint;
  const headers = {
    'apikey': cfg.supabaseKey,
    'Authorization': 'Bearer ' + cfg.supabaseKey,
    'Content-Type': 'application/json'
  };
  if (preferHeader) {
    headers['Prefer'] = preferHeader;
  }
  const options = {
    method: method || 'get',
    headers: headers,
    muteHttpExceptions: true
  };
  if (bodyObj) {
    options.payload = JSON.stringify(bodyObj);
  }
  const res = UrlFetchApp.fetch(url, options);
  const code = res.getResponseCode();
  const text = res.getContentText();
  if (code >= 400) {
    throw new Error('Supabase Error (' + code + '): ' + text);
  }
  return text ? JSON.parse(text) : {};
}

/**
 * Handler GET: Cek status koneksi & konfigurasi server GAS
 */
function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'health';
  if (action === 'health') {
    return jsonResponse_({
      success: true,
      service: 'Sipustadig GAS Backend v3.0',
      timestamp: new Date().toISOString()
    });
  }
  return jsonResponse_({ success: false, message: 'Action GET tidak dikenali.' });
}

/**
 * Handler POST Utama:
 * Menerima request dari Frontend Vercel dengan Content-Type: text/plain;charset=utf-8
 * agar bebas kendala preflight CORS di browser.
 */
function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse_({ success: false, message: 'Payload kosong.' });
    }
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    switch (action) {
      case 'uploadCoverOrFile':
        return handleUploadCoverOrFile_(payload);
      case 'importMembersExcel':
        return handleImportMembers_(payload);
      case 'importBooksExcel':
        return handleImportBooks_(payload);
      case 'exportReportSheet':
        return handleExportReportSheet_(payload);
      case 'generateLabelsPdf':
        return handleGenerateLabelsPdf_(payload);
      case 'dailyOverdueCron':
        return jsonResponse_(dailyOverdueCron());
      default:
        return jsonResponse_({ success: false, message: 'Action POST tidak dikenali: ' + action });
    }
  } catch (err) {
    return jsonResponse_({
      success: false,
      message: err.message || String(err)
    });
  }
}

/**
 * 1. Upload Cover Buku (maks 2MB) atau E-Book PDF ke Google Drive Sekolah
 */
function handleUploadCoverOrFile_(payload) {
  const cfg = getConfig_();
  const folder = cfg.driveFolderId
    ? DriveApp.getFolderById(cfg.driveFolderId)
    : DriveApp.getRootFolder();

  const decoded = Utilities.base64Decode(payload.base64Data);
  const mimeType = payload.mimeType || 'image/jpeg';
  const fileName = payload.fileName || ('cover_' + new Date().getTime() + '.jpg');

  const blob = Utilities.newBlob(decoded, mimeType, fileName);
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

  const fileId = file.getId();
  const publicUrl = 'https://drive.google.com/uc?export=view&id=' + fileId;

  return jsonResponse_({
    success: true,
    message: 'File berhasil diunggah ke Google Drive sekolah.',
    data: {
      drive_file_id: fileId,
      cover_path: publicUrl,
      view_url: file.getUrl()
    }
  });
}

/**
 * 2. Bulk Import Anggota dari Data Excel / Google Sheets ke Supabase
 */
function handleImportMembers_(payload) {
  const rows = payload.rows || [];
  if (!rows.length) {
    return jsonResponse_({ success: false, message: 'Data baris import anggota kosong.' });
  }
  const inserted = supabaseFetch_('members', 'post', rows, 'return=representation,resolution=merge-duplicates');
  return jsonResponse_({
    success: true,
    message: 'Import anggota selesai.',
    data: {
      total_rows: rows.length,
      success_rows: Array.isArray(inserted) ? inserted.length : rows.length,
      failed_rows: 0
    }
  });
}

/**
 * 4. Export Laporan Sirkulasi / Denda / Anggota ke Google Sheets & Link Unduh .xlsx
 */
function handleExportReportSheet_(payload) {
  const cfg = getConfig_();
  const ss = cfg.reportSheetId
    ? SpreadsheetApp.openById(cfg.reportSheetId)
    : SpreadsheetApp.create('Laporan_Perpustakaan_Sipustadig_' + Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyyMMdd'));

  const sheetName = payload.sheetName || 'Laporan Sirkulasi';
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  } else {
    sheet.clear();
  }

  const headers = payload.headers || ['No. Transaksi', 'Nomor Anggota', 'Nama Anggota', 'Kode Eksemplar', 'Judul Buku', 'Jatuh Tempo', 'Status'];
  const rows = payload.rows || [];

  sheet.appendRow(headers);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#eef2f6');

  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
  sheet.autoResizeColumns(1, headers.length);

  const fileId = ss.getId();
  return jsonResponse_({
    success: true,
    message: 'Laporan berhasil diekspor ke Google Sheets.',
    data: {
      spreadsheet_url: ss.getUrl(),
      xlsx_download_url: 'https://docs.google.com/spreadsheets/d/' + fileId + '/export?format=xlsx',
      pdf_download_url: 'https://docs.google.com/spreadsheets/d/' + fileId + '/export?format=pdf'
    }
  });
}

/**
 * 6. CRON JOB HARIAN (Pasang pada Triggers > Time-driven > Day timer > 07:00 - 08:00)
 * - Mengecek pinjaman melewati due_at di Supabase dan mengubah status menjadi 'overdue'
 * - Sekaligus menjaga proyek Supabase Free Tier agar tetap aktif (anti-pause 7 hari)
 */
function dailyOverdueCron() {
  const today = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd');
  const overdueItems = supabaseFetch_(
    'loan_items?status=eq.borrowed&due_at=lt.' + today + '&select=id,loan_transaction_id,due_at',
    'get'
  );

  if (Array.isArray(overdueItems) && overdueItems.length > 0) {
    supabaseFetch_(
      'loan_items?status=eq.borrowed&due_at=lt.' + today,
      'patch',
      { status: 'overdue', updated_at: new Date().toISOString() }
    );
  }

  return {
    success: true,
    checked_date: today,
    marked_overdue_count: Array.isArray(overdueItems) ? overdueItems.length : 0
  };
}
